async function bal(pub) { const r = await (await fetch(`${HORIZON}/accounts/${pub}`)).json(); const b = r.balances.find((x) => x.asset_type === 'native').balance; const [w, f = ''] = b.split('.'); return (BigInt(w) * 10_000_000n + BigInt((f + '0000000').slice(0, 7))).toString(); }
async function horizonHas(hash) { for (let i = 0; i < 30; i++) { const r = await fetch(`${HORIZON}/transactions/${hash}`); if (r.status === 200) return true; await sleep(2000); } return false; }

const withRF = (tx, r, rf) => s.TransactionBuilder.cloneFrom(s.rpc.assembleTransaction(tx, r).build(), { fee: String(INCL), sorobanData: r.transactionData.setResourceFee(BigInt(rf)).build() }).build();

async function submit(label, tx, k, expect, extra) {
  tx.sign(k);
  const row = { label, ...extra, envelopeFee: tx.fee, inclusionFee: INCL };
  row.balanceBefore = await bal(k.publicKey());
  const send = await server.sendTransaction(tx);
  row.sendStatus = send.status;
  if (send.status !== 'PENDING') {
    row.hash = null; row.refusedAtSubmission = true;
    try { row.errorResult = JSON.stringify(send.errorResult, (x, v) => (typeof v === 'bigint' ? v.toString() : v)).slice(0, 400); } catch { row.errorResult = 'undetermined'; }
    await sleep(6000); row.balanceAfter = await bal(k.publicKey()); row.balanceDelta = (BigInt(row.balanceAfter) - BigInt(row.balanceBefore)).toString();
    results.push(row); save(); console.log(label, 'REFUSED at submission', send.status, row.errorResult);
    if (expect !== 'refused-or-failed') throw new Error('unexpected refusal at ' + label);
    return row;
  }
  row.hash = send.hash;
  let g; for (let i = 0; i < 60; i++) { await sleep(2000); g = await server.getTransaction(send.hash); if (g.status !== 'NOT_FOUND') break; }
  row.status = g.status; row.ledger = g.ledger;
  const fee = cli(['tx', 'fetch', 'fee', '--hash', send.hash, '--network', 'testnet', '--output', 'json']);
  row.feeInfo = fee; row.feeCharged = String(fee.charged.fee);
  const meta = execFileSync('stellar', ['tx', 'fetch', 'meta', '--hash', send.hash, '--network', 'testnet', '--output', 'json'], { encoding: 'utf8' });
  const m = meta.match(/\{"ext":"v0","total_non_refundable_resource_fee_charged":"(\d+)","total_refundable_resource_fee_charged":"(\d+)","rent_fee_charged":"(\d+)"\}/);
  row.meta = m ? { totalNonRefundableResourceFeeCharged: m[1], totalRefundableResourceFeeCharged: m[2], rentFeeCharged: m[3] } : 'undetermined';
  if (g.status === 'FAILED') { const r = cli(['tx', 'fetch', 'result', '--hash', send.hash, '--network', 'testnet', '--output', 'json']); row.result = JSON.stringify(r).slice(0, 600); }
  await horizonHas(send.hash); row.balanceAfter = await bal(k.publicKey());
  row.balanceDelta = (BigInt(row.balanceAfter) - BigInt(row.balanceBefore)).toString();
  row.notCharged = (BigInt(tx.fee) - BigInt(row.feeCharged)).toString();
  results.push(row); save();
  console.log(label, row.status, row.hash, 'env', tx.fee, 'declRF', extra.declaredResourceFee, 'charged', row.feeCharged, 'delta', row.balanceDelta, 'meta', JSON.stringify(row.meta));
  if (expect === 'success' && g.status !== 'SUCCESS') throw new Error('unexpected failure at ' + label);
  if (expect === 'failed' && g.status !== 'FAILED') throw new Error(`expected failure at ${label}, got ${g.status}`);
  return row;
}
