async function rpc(method,params={},name) {
  const response=await fetch(url,{method:'POST',headers:{'Content-Type':'application/json','User-Agent':'evergreen-w1-testnet-setup'},body:JSON.stringify({jsonrpc:'2.0',id:1,method,params}),signal:AbortSignal.timeout(25000)});
  const raw=await response.text();
  if(name) fs.writeFileSync(`${out}/${name}.json`,raw,{flag:'wx'});
  if(!response.ok) throw Error(`RPC HTTP ${response.status}`);
  const body=JSON.parse(raw);if(body.error) throw Error(JSON.stringify(body.error));
  return {body,raw};
}

async function submit(operation,name,soroban=true) {
  if(fs.existsSync(`${out}/${name}-send.json`)) throw Error('Submission already recorded; reconcile hash before retrying');
  const account=await server.getAccount(dev.publicKey());
  const unsigned=new s.TransactionBuilder(account,{fee:'1000',networkPassphrase:passphrase}).addOperation(operation).setTimeout(300).build();
  let prepared=unsigned, sim;
  if(soroban) {
    sim=(await rpc('simulateTransaction',{transaction:unsigned.toXDR()},`${name}-simulation-${Date.now()}`)).body.result;
    if(sim.error) throw Error(`Simulation failed: ${sim.error}`);
    prepared=s.rpc.assembleTransaction(unsigned,sim).build();
  }
  if(!live) {console.log('DRY RUN: simulation/build succeeded; no transaction submitted.');return {sim};}
  prepared.sign(dev);
  const hash=Buffer.from(prepared.hash()).toString('hex');
  save(`${name}-intent`,{hash,at:new Date().toISOString(),signer:dev.publicKey(),network:'Testnet'});
  const result=(await rpc('sendTransaction',{transaction:prepared.toXDR()},`${name}-send`)).body.result;
  console.log(JSON.stringify({name,hash,status:result.status}));
  if(!['PENDING','DUPLICATE'].includes(result.status)) throw Error(`Submission ${result.status}: ${result.errorResultXdr??''}`);
  for(let i=0;i<40;i++) {
    await delay(2000);
    const got=await rpc('getTransaction',{hash});
    if(got.body.result.status==='NOT_FOUND')continue;
    fs.writeFileSync(`${out}/${name}-transaction.json`,got.raw,{flag:'wx'});
    if(got.body.result.status!=='SUCCESS') throw Error(`Transaction ${got.body.result.status}`);
    console.log(JSON.stringify({name,hash,status:'SUCCESS',ledger:got.body.result.ledger}));
    return {sim,result:got.body.result,hash};
  }
  throw Error('Confirmation pending: query the recorded hash, do not resubmit');
}
